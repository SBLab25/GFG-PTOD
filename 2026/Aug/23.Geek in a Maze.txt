class Solution {
    class Pair {
        int row, col, curup, curdown;
        Pair(int row, int col, int curup, int curdown) {
            this.row = row;
            this.col = col;
            this.curup = curup;
            this.curdown = curdown;
        }
    }
    public int numberOfCells(int r, int c, int u, int d, char[][] mat) {
        // code here
        int n = mat.length;
        int m = mat[0].length;

        if (mat[r][c] == '#') return 0;

        int[][] minUp = new int[n][m];
        for (int i = 0; i < n; i++) {
            Arrays.fill(minUp[i], Integer.MAX_VALUE);
        }

        int[] dr = {-1, 0, 1, 0};
        int[] dc = {0, -1, 0, 1};
        PriorityQueue<Pair> pq = new PriorityQueue<>((a, b) -> Integer.compare(a.curup, b.curup));
        minUp[r][c] = 0;
        pq.add(new Pair(r, c, 0, 0));

        int reachableCount = 0;
        boolean[][] visited = new boolean[n][m];

        while (!pq.isEmpty()) {
            Pair it = pq.poll();
            int ro = it.row;
            int co = it.col;
            int cup = it.curup;
            int cd = it.curdown;

            if (visited[ro][co]) continue;
            visited[ro][co] = true;
            reachableCount++; 

            for (int i = 0; i < 4; i++) {
                int nr = ro + dr[i];
                int nc = co + dc[i];

                if (nr >= 0 && nr < n && nc >= 0 && nc < m && mat[nr][nc] != '#') {
                    int nextUp = cup;
                    int nextDown = cd;

                    if (dr[i] == -1) {
                        nextUp++;
                    } else if (dr[i] == 1) {
                        nextDown++; 
                    }
                    if (nextUp <= u && nextDown <= d) {
                        if (nextUp < minUp[nr][nc]) {
                            minUp[nr][nc] = nextUp;
                            pq.add(new Pair(nr, nc, nextUp, nextDown));
                        }
                    }
                }
            }
        }

        return reachableCount;
    }
}
